\begin{tikzpicture}[
  font=\footnotesize,
  sq/.style={draw=black!35, minimum size=6mm, inner sep=0pt},
  ain/.style={sq, fill=laneA!25, draw=laneA!60!black, thick},
  bin/.style={sq, fill=laneB!22, draw=laneB!65!black, thick},
  rl/.style={font=\scriptsize, text=black!70, align=right},
  cl/.style={font=\scriptsize, text=probe!40!black, align=center},
  cmpbox/.style={draw=probe!55!black, dashed, thick, rounded corners=2pt, inner sep=2.5pt},
]
  \def\dx{0.72}
  % Row 1 (top): experiment 1 -- a placed at position j (slot 5)
  \node[rl, anchor=east] at (-0.35,2.4) {experiment 1:\\ run $a$ at $j$};
  \foreach \k in {0,...,6}{\ifnum\k=5 \node[ain](aj) at({\k*\dx},2.4){$a$};\else\node[sq]at({\k*\dx},2.4){};\fi}
  % Row 2 (middle): the original run -- a at i (slot 2), b at j (slot 5)
  \node[rl, anchor=east] at (-0.35,1.2) {original run};
  \foreach \k in {0,...,6}{\ifnum\k=2 \node[ain](a0)at({\k*\dx},1.2){$a$};\else\ifnum\k=5 \node[bin](b0)at({\k*\dx},1.2){$b$};\else\node[sq]at({\k*\dx},1.2){};\fi\fi}
  % Row 3 (bottom): experiment 2 -- b placed at position i (slot 2)
  \node[rl, anchor=east] at (-0.35,0) {experiment 2:\\ run $b$ at $i$};
  \foreach \k in {0,...,6}{\ifnum\k=2 \node[bin](bi)at({\k*\dx},0){$b$};\else\node[sq]at({\k*\dx},0){};\fi}
  % each dashed box is one fixed context: compare a and b inside it
  \node[cmpbox, fit=(aj)(b0)] (boxj) {};
  \node[cmpbox, fit=(a0)(bi)] (boxi) {};
  \node[cl, anchor=south] at (boxj.north) {context $j$: $a@j = b@j$?};
  \node[cl, anchor=north] at (boxi.south) {context $i$: $a@i = b@i$?};
  % the candidate sits on the original run
  \node[font=\scriptsize, text=culprit!72!black, align=left, anchor=west] at ({6*\dx+0.35},1.2)
     {candidate:\\ $a@i \neq b@j$};
  % the decision, as the overall conclusion
  \node[font=\scriptsize, text=black!78, align=center, anchor=north] at ({3*\dx},-1.3)
     {$a,b$ agree in both contexts: false positive.\qquad $a,b$ differ in either: violation.};
\end{tikzpicture}
