% ---- notation list: show each abbreviation's expansion once ----
% The class's \formatglossarydescription prints "long (description)" for any entry
% that has a long form. Plain acronyms (long-short style) get description = long by
% default, so the list rendered the expansion twice, as "long (long)". Collapse it:
% an abbreviation shows its expansion once; a notation symbol (no long field) still
% shows its description. Guarded so a noabbrevs build (macro undefined) stays safe.
\makeatletter
\@ifundefined{formatglossarydescription}{}{%
  % Show each meaning once, with its first letter capitalised in the list.
  \renewcommand*{\formatglossarydescription}[1]{%
    \ifglsfieldeq{#1}{long}{}{%
      \Glossentrydesc{#1}%
    }{%
      \Glsentrylong{#1}%
    }%
  }%
  % Widen the name column so a multi-word term such as "speculation contract"
  % stays on one line, and drop the page-number column (a page reference showed
  % only against the few abbreviations referenced in the body).
  \renewglossarystyle{iitthesis}{%
    \renewenvironment{theglossary}%
      {\begin{longtable}{@{}p{4cm}@{\quad}p{\textwidth - 4cm - 1em}@{}}}%
      {\end{longtable}}%
    \renewcommand*{\glossaryheader}{}%
    \renewcommand*{\glsgroupheading}[1]{}%
    \renewcommand*{\glossentry}[2]{%
      \textbf{\glstarget{##1}{\glossentryname{##1}}}%
      & \formatglossarydescription{##1}%
      \tabularnewline%
    }%
    \renewcommand*{\glsgroupskip}{}%
  }%
}
\makeatother
